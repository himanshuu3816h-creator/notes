################################CLIENT

sudo apt install puppet-agent -y


sudo nano /etc/puppet/puppet.conf

[main]
server = pserver

sudo systemctl enable puppet
sudo systemctl start puppet


sudo puppet config print server
output - pserver


Command	Purpose
puppet --version	                               Check version
sudo systemctl start puppetserver	               Start server
sudo systemctl start puppet	                     Start agent
sudo systemctl stop puppet	                     Stop agent
sudo puppet agent -t	                           Run agent once
sudo puppetserver ca list 	                     List pending certificates
sudo puppetserver ca sign --certname agent-1	   Approve agent
sudo systemctl status puppetserver	             Check server
